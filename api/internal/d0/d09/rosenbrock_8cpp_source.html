100  void Rosenbrock::set_iteration_matrix() {
101  // Iteration matrix M/(h gamma) - J of the nominal [x; z]
102  sp_iter_ = blockcat(sp_jac_[IVP_ODE_X] + Sparsity::diag(nx1_),
103  sp_jac_[IVP_ODE_Z], sp_jac_[IVP_ALG_X], sp_jac_[IVP_ALG_Z]);
104  // Positions of the nonzeros of the blocks ODE_X, ALG_X, ODE_Z, ALG_Z, then the x-diagonal
105  iter_map_.clear();
106  const casadi_int blk[4] = {IVP_ODE_X, IVP_ALG_X, IVP_ODE_Z, IVP_ALG_Z};
107  for (casadi_int k = 0; k < 4; ++k) {
108  std::vector<casadi_int> r = sp_jac_[blk[k]].get_row(), c = sp_jac_[blk[k]].get_col();
109  casadi_int r_off = blk[k] == IVP_ALG_X || blk[k] == IVP_ALG_Z ? nx1_ : 0;
110  casadi_int c_off = blk[k] == IVP_ODE_Z || blk[k] == IVP_ALG_Z ? nx1_ : 0;
111  for (size_t i = 0; i < r.size(); ++i) {
112  iter_map_.push_back(sp_iter_.get_nz(r[i] + r_off, c[i] + c_off));
113  }
114  }
115  for (casadi_int i = 0; i < nx1_; ++i) iter_map_.push_back(sp_iter_.get_nz(i, i));
116  // Symbolic sparse QR factorizations
117  sp_iter_.qr_sparse(sp_iter_v_, sp_iter_r_, iter_prinv_, iter_pc_);
118  if (nz1_ > 0) sp_jac_[IVP_ALG_Z].qr_sparse(sp_alg_v_, sp_alg_r_, alg_prinv_, alg_pc_);
119  }

121  void Rosenbrock::set_coefficients() {
122  // Coefficients in original form (rosenbrock_tableau.hpp), Gamma lower triangular
123  using namespace rosenbrock_tableau;
124  const casadi_int s = n_stages;
125  // Inverse of Gamma = (gamma_ij), lower triangular, in long double
126  std::vector<long double> Gi(s * s, 0);
127  for (casadi_int j = 0; j < s; ++j) {
128  Gi[j * s + j] = 1 / gamma;
129  for (casadi_int i = j + 1; i < s; ++i) {
130  long double acc = 0;
131  for (casadi_int k = j; k < i; ++k) acc += gamma_ij[i][k] * Gi[k * s + j];
132  Gi[i * s + j] = -acc / gamma;
133  }
134  }
135  // a = A Gamma^-1, c = diag(1/gamma) - Gamma^-1, m = Gamma^-T b, m_err = Gamma^-T (b - bhat)
136  a_.assign(s * s, 0);
137  c_.assign(s * s, 0);
138  alpha_.assign(s, 0);
139  gamma_sum_.assign(s, 0);
140  m_.assign(s, 0);
141  m_err_.assign(s, 0);
142  for (casadi_int i = 0; i < s; ++i) {
143  long double alpha_i = 0, gamma_i = gamma;
144  for (casadi_int j = 0; j < i; ++j) {
145  alpha_i += alpha[i][j];
146  gamma_i += gamma_ij[i][j];
147  long double aij = 0;
148  for (casadi_int k = j; k < i; ++k) aij += alpha[i][k] * Gi[k * s + j];
149  a_[i * s + j] = static_cast<double>(aij);
150  c_[i * s + j] = static_cast<double>(-Gi[i * s + j]);
151  }
152  alpha_[i] = static_cast<double>(alpha_i);
153  gamma_sum_[i] = static_cast<double>(gamma_i);
154  long double mi = 0, mhi = 0;
155  for (casadi_int k = i; k < s; ++k) {
156  mi += b[k] * Gi[k * s + i];
157  mhi += bhat[k] * Gi[k * s + i];
158  }
159  m_[i] = static_cast<double>(mi);
160  m_err_[i] = static_cast<double>(mi - mhi);
161  }
162  order_err_ = order_embedded;
163  // Dense output weights in the same form: m(theta) = Gamma^-T b(theta)
164  n_dense_ = n_dense;
165  m_dense_.assign(n_dense * s, 0);
166  for (casadi_int q = 0; q < n_dense; ++q) {
167  for (casadi_int i = 0; i < s; ++i) {
168  long double mi = 0;
169  for (casadi_int k = i; k < s; ++k) mi += b_dense[q][k] * Gi[k * s + i];
170  m_dense_[q * s + i] = static_cast<double>(mi);
171  }
172  }
173  }

175  void Rosenbrock::set_rosenbrock_prob() {
176  // Problem structure of the method
177  pr_.ivp = &p_;
178  pr_.gamma = static_cast<double>(rosenbrock_tableau::gamma);
179  pr_.c = get_ptr(c_);
180  pr_.gamma_sum = get_ptr(gamma_sum_);
181  pr_.sp_iter = sp_iter_;
182  pr_.iter_map = get_ptr(iter_map_);
183  pr_.sp_iter_v = sp_iter_v_;
184  pr_.sp_iter_r = sp_iter_r_;
185  pr_.iter_prinv = get_ptr(iter_prinv_);
186  pr_.iter_pc = get_ptr(iter_pc_);
187  pr_.sp_alg_v = nz1_ > 0 ? static_cast<const casadi_int*>(sp_alg_v_) : nullptr;
188  pr_.sp_alg_r = nz1_ > 0 ? static_cast<const casadi_int*>(sp_alg_r_) : nullptr;
189  pr_.alg_prinv = get_ptr(alg_prinv_);
190  pr_.alg_pc = get_ptr(alg_pc_);
191  pr_.init_tol = init_tol_;
192  pr_.init_max_iter = init_max_iter_;
193  casadi_rosenbrock_setup(&pr_);
194  // Method of the original integrator, if its steps are replayed
195  if (replay_) {
196  pr_nom_ = static_cast<const Rosenbrock*>(root())->pr_;
197  pr_nom_.ivp = &p_nom_;
198  casadi_rosenbrock_setup(&pr_nom_);
199  }
200  }

203  casadi_int*& iw, double*& w) const {

209  m->r_nom.prob = &pr_nom_;

211  m->r.nom = &m->r_nom;
212  }

220  void Rosenbrock::codegen_method(CodeGenerator& g) const {
221  g.add_auxiliary(CodeGenerator::AUX_ROSENBROCK);
222  codegen_rosenbrock_prob(g, "pe", "p");

227  g << "en.prob = &pen;\n";

233  void Rosenbrock::codegen_rosenbrock_prob(CodeGenerator& g, const std::string& pe,
234  const std::string& p) const {
235  // Problem structure of the method in variable pe, as set_rosenbrock_prob
236  auto set = [&](const std::string& field, const std::string& value) -> void {
237  g << pe << "." << field << " = " << value << ";\n";
238  };
239  set("ivp", "&" + p);
240  set("gamma", g.constant(pr_.gamma));
241  set("c", g.constant(c_));
242  set("gamma_sum", g.constant(gamma_sum_));
243  set("sp_iter", g.sparsity(sp_iter_));
244  set("iter_map", g.constant(iter_map_));
245  set("sp_iter_v", g.sparsity(sp_iter_v_));
246  set("sp_iter_r", g.sparsity(sp_iter_r_));
247  set("iter_prinv", g.constant(iter_prinv_));
248  set("iter_pc", g.constant(iter_pc_));
249  set("sp_alg_v", nz1_ > 0 ? g.sparsity(sp_alg_v_) : "0");
250  set("sp_alg_r", nz1_ > 0 ? g.sparsity(sp_alg_r_) : "0");
251  set("alg_prinv", nz1_ > 0 ? g.constant(alg_prinv_) : "0");
252  set("alg_pc", nz1_ > 0 ? g.constant(alg_pc_) : "0");
253  set("init_tol", g.constant(pr_.init_tol));
254  set("init_max_iter", str(pr_.init_max_iter));
255  g << "casadi_rosenbrock_setup(&" << pe << ");\n";
256  }
