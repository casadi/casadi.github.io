238  for (casadi_int k=0; k<nout(); ++k) {
239  bvec_t* v = res[k];
240  if (!v) continue;
241  for (casadi_int i=0; i<sparsity(k).nnz(); ++i) v[i] = ~static_cast<bvec_t>(0);
242  }
243  return 0;
244  }
