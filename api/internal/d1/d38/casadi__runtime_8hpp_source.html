356  #define CASADI_GEMM_NT(M, N, K, A, LDA, B, LDB, C, LDC) \
357  for (i=0, rr=C; i<M; ++i) \
358  for (j=0; j<N; ++j, ++rr) \
359  for (k=0, ss=A+i*LDA, tt=B+j*LDB; k<K; ++k) \
360  *rr += *ss++**tt++;
