108 // SYMBOL "ivp_prob"
109 template<typename T1>
110 struct casadi_ivp_prob {
111  // Nominal block sizes: differential, algebraic, quadrature states, parameters, controls
112  casadi_int nx1, nz1, nq1, np1, nu1;
113  // Number of forward and adjoint directions
114  casadi_int nfwd, nadj;
115  // Stages: increments U_i, arguments y0 + sum_j a_ij U_j at t0 + alpha_i h (a strictly lower
116  // triangular, row-major), solution y1 = y0 + sum_i m_i U_i
117  casadi_int n_stages;
118  const T1 *a, *alpha, *m;
119  // Dense output weights m_i(theta) = sum_q m_dense[q * n_stages + i] theta^(q+1)
120  casadi_int n_dense;
121  const T1* m_dense;
122  // Sparsity patterns of the Jacobian blocks (null: not used), see casadi_ivp_jac_t
123  const casadi_int* sp_jac[IVP_NUM_JAC];
124  // Capacity of the tape in the work vector (0: the caller provides the tape)
125  casadi_int tape_size;
126  // Fixed steps: each run of steps up to a stop has equal steps of at most step_size
127  // (0: adaptive steps)
128  T1 step_size;
129  // Error weights: leading entries of x, z, q subject to error control, tolerances
130  casadi_int nx_err, nz_err, nq_err;
131  T1 abstol, reltol;
132  // Adaptive steps: order and weights of the error estimate sum_i m_err_i U_i, step size
133  // controller, maximum number of steps per output interval
134  casadi_int order_err;
135  const T1* m_err;
136  T1 safety, fac_min, fac_max, h_min, h_max, h_init;
137  casadi_int max_num_steps;
138  // Zero-crossing events: number, transition given, tolerances on the event time (acceptable:
139  // negative if none), iterations to locate an event, events per evaluation
140  casadi_int ne;
141  int has_transition;
142  T1 event_tol, event_acceptable_tol;
143  casadi_int max_event_iter, max_events;
144  // Derived: forward and adjoint state dimensions, controls, vector lengths
145  casadi_int nx, nz, nq, nu, nrx, nrz, nrq, nrp, nru, ny, nr;
146  // Derived: offsets of the Jacobian blocks in a buffer with one direction, total nonzeros
147  casadi_int jac_off[IVP_NUM_JAC + 1];
148  casadi_int nnz_jac;
149 };

153 inline
154 casadi_int casadi_ivp_nnz(const casadi_int* sp) {
155  // Number of nonzeros of a sparsity pattern, zero if null
156  return sp ? sp[2 + sp[1]] : 0;
157 }

159 // SYMBOL "ivp_setup"
160 template<typename T1>
161 void casadi_ivp_setup(casadi_ivp_prob<T1>* p) {
162  casadi_int k;
163  // Forward states including sensitivities
164  p->nx = p->nx1 * (1 + p->nfwd);
165  p->nz = p->nz1 * (1 + p->nfwd);
166  p->nq = p->nq1 * (1 + p->nfwd);
167  p->nu = p->nu1 * (1 + p->nfwd);
168  // Adjoint states and adjoint parameters, controls, including forward sensitivities
169  p->nrx = p->nx1 * p->nadj * (1 + p->nfwd);
170  p->nrz = p->nz1 * p->nadj * (1 + p->nfwd);
171  p->nrq = p->nq1 * p->nadj * (1 + p->nfwd);
172  p->nrp = p->np1 * p->nadj * (1 + p->nfwd);
173  p->nru = p->nu1 * p->nadj * (1 + p->nfwd);
174  p->ny = p->nx + p->nz + p->nq;
175  p->nr = p->nrx + p->nrz + p->nrq;
176  // Jacobian blocks
177  p->jac_off[0] = 0;
178  for (k = 0; k < IVP_NUM_JAC; ++k) {
179  p->jac_off[k + 1] = p->jac_off[k] + casadi_ivp_nnz(p->sp_jac[k]);
180  }
181  p->nnz_jac = p->jac_off[IVP_NUM_JAC];
182 }

186 template<typename T1>
187 struct casadi_ivp_data {
188  // Problem structure
189  const casadi_ivp_prob<T1>* prob;
190  // Data whose task is requested: this one, or one the method delegates to
191  casadi_ivp_data<T1>* active;
192  // Nominal integration whose steps are replayed (or null), stop after every accepted step
193  casadi_ivp_data<T1>* nom;
194  int pause;
195  // Task for the caller, where to resume (casadi_ivp_entry_t or a phase of the method), status
196  casadi_ivp_task_t task;
197  // Step size control requested, step accepted by it
198  casadi_ivp_control_t control;
199  int accepted;
200  casadi_int next;
201  casadi_ivp_flag_t status;
202  // Set by the caller if a task could not be carried out
203  int fail;
204  // Task arguments and results
205  T1 t_arg;
206  const T1 *y_arg, *w;
207  T1 *res;
208  // Current time and state, output time, latest time to step to, start of a retreat
209  T1 t, t_out, t_stop, t_start;
210  T1* y;
211  // State at the output time
212  T1* y_out;
213  // Controls set by the caller, controls of the current or popped step
214  const T1* u;
215  T1* u_step;
216  // Current step: start time, size, start state, stage counter
217  T1 t0, h;
218  // End time and end of the output interval of a popped record
219  T1 t1, t_record;
220  T1* y0;
221  casadi_int stage;
222  // Stage increments U_i (n_stages x ny), stage argument, dense output weights
223  T1 *U, *y_stage, *w_dense;
224  // Adjoint state, stage adjoints (n_stages x nr), adjoint of the start of the popped step
225  T1 *ry, *rU, *ry0;
226  // Adjoint seed of an output at time t_seed
227  T1* seed;
228  T1 t_seed;
229  // Control adjoints of the popped step from outputs inside it
230  T1* ru_step;
231  // Jacobian blocks and their forward derivatives, nonzeros in casadi_ivp_jac_t order
232  T1 *jac, *jac_fwd;
233  // Adjoint seeds of the Jacobian blocks, with forward derivatives
234  T1 *jac_adj, *jac_adj_fwd;
235  // Results of EVAL_VJP and EVAL_JAC_ADJ: adjoints of x, z, p, u
236  T1 *vjp_x, *vjp_z, *vjp_p, *vjp_u;
237  // Accumulated adjoints of the parameters and controls
238  T1 *adj_p, *adj_u;
239  // Tape: records [U (steps only), y0, u, h, t0, t1, t_out, type]; entries used, capacity, needed
240  T1* tape;
241  casadi_int tape_pos, tape_cap, tape_need;
242  // Reset pending, taping requested
243  int reset, taping;
244  // Record type, popped record that belongs to an earlier interval
245  casadi_ivp_record_t record;
246  int record_pending;
247  // Fixed steps: current run start, stop, step size, number of steps, steps taken
248  T1 run_t0, run_t1, run_h;
249  casadi_int run_n, run_k;
250  // Adaptive steps: proposed next step size, previous error, last step rejected, clipped at
251  // the stop or ending at the stop
252  T1 h_next, err_prev;
253  int last_rejected, last_clipped, last_step;
254  // Steps taken in the current output interval
255  casadi_int n_steps_interval;
256  // Events: zero-crossing functions at the bracket ends (e0 also at the step start) and at a
257  // trial time, candidates, time derivatives at these, selection of the event, state derivative
258  T1 *e0, *e1, *em, *e_mask, *edot0, *edot1, *edot, *ev_w, *ydot;
259  // Event phase, e0 known at t, state changed by an event (algebraic states inconsistent),
260  // restart after an event
261  int ev_phase, e_valid, jump, ev_restart;
262  // Bracket, trial time, event time and its accuracy, the constant 1
263  T1 ev_ta, ev_tb, ev_tm, t_ev, ev_dt, ev_one;
264  // Iterations, event in the last step (-1: none), events in this evaluation
265  casadi_int ev_iter, ev_fired, n_events;
266  // Statistics
267  casadi_int n_steps, n_reject, n_rhs, n_jac, n_vjp;
268 };

270 // SYMBOL "ivp_work"
271 template<typename T1>
272 void casadi_ivp_work(const casadi_ivp_prob<T1>* p, casadi_int* sz_arg,
273  casadi_int* sz_res, casadi_int* sz_iw, casadi_int* sz_w) {
274  // State: y, y_out, y0, stages, stage argument, dense output weights
275  *sz_w += p->ny * (p->n_stages + 4) + p->n_stages;
276  // Adjoint: adjoint state, stage adjoints, adjoint of the step start, seed
277  *sz_w += p->nr * (p->n_stages + 3);
278  // Controls of a step, their adjoints from outputs inside it
279  *sz_w += p->nu + p->nru;
280  // Jacobian blocks, forward derivatives, adjoint seeds and their forward derivatives
281  *sz_w += p->nnz_jac * (1 + p->nfwd) * (1 + p->nadj);
282  // Function adjoints, accumulated adjoints
283  *sz_w += p->nrx + p->nrz + 2 * (p->nrp + p->nru);
284  // Tape
285  *sz_w += p->tape_size;
286  // Events
287  if (p->ne > 0) *sz_w += 8 * p->ne + p->ny;
288 }

290 // SYMBOL "ivp_set_work"
291 template<typename T1>
292 void casadi_ivp_set_work(casadi_ivp_data<T1>* d, const T1*** arg, T1*** res,
293  casadi_int** iw, T1** w) {
294  const casadi_ivp_prob<T1>* p = d->prob;
295  d->y = *w; *w += p->ny;
296  d->y_out = *w; *w += p->ny;
297  d->y0 = *w; *w += p->ny;
298  d->U = *w; *w += p->ny * p->n_stages;
299  d->y_stage = *w; *w += p->ny;
300  d->w_dense = *w; *w += p->n_stages;
301  d->ry = *w; *w += p->nr;
302  d->rU = *w; *w += p->nr * p->n_stages;
303  d->ry0 = *w; *w += p->nr;
304  d->seed = *w; *w += p->nr;
305  d->u_step = *w; *w += p->nu;
306  d->ru_step = *w; *w += p->nru;
307  d->jac = *w; *w += p->nnz_jac;
308  d->jac_fwd = *w; *w += p->nnz_jac * p->nfwd;
309  d->jac_adj = *w; *w += p->nnz_jac * p->nadj;
310  d->jac_adj_fwd = *w; *w += p->nnz_jac * p->nadj * p->nfwd;
311  d->vjp_x = *w; *w += p->nrx;
312  d->vjp_z = *w; *w += p->nrz;
313  d->vjp_p = *w; *w += p->nrp;
314  d->vjp_u = *w; *w += p->nru;
315  d->adj_p = *w; *w += p->nrp;
316  d->adj_u = *w; *w += p->nru;
317  d->tape = *w; *w += p->tape_size;
318  if (p->ne > 0) {
319  d->e0 = *w; *w += p->ne;
320  d->e1 = *w; *w += p->ne;
321  d->em = *w; *w += p->ne;
322  d->e_mask = *w; *w += p->ne;
323  d->edot0 = *w; *w += p->ne;
324  d->edot1 = *w; *w += p->ne;
325  d->edot = *w; *w += p->ne;
326  d->ev_w = *w; *w += p->ne;
327  d->ydot = *w; *w += p->ny;
328  }
329  d->tape_cap = p->tape_size;
330  d->tape_pos = 0;
331  d->next = IVP_DONE;
332  d->status = IVP_SUCCESS;
333  d->fail = 0;
334  d->reset = 0;
335  d->taping = 0;
336  d->record_pending = 0;
337  d->active = d;
338  d->nom = 0;
339  d->pause = 0;
340  d->ev_phase = 0;
341  d->e_valid = 0;
342  d->jump = 0;
343  d->ev_restart = 0;
344  d->ev_dt = 0;
345  d->ev_one = 1;
346  d->ev_fired = -1;
347  d->n_events = 0;
348 }

356 // SYMBOL "ivp_reset"
357 template<typename T1>
358 void casadi_ivp_reset(casadi_ivp_data<T1>* d, T1 t, const T1* u, int first_call) {
359  // Start at time t, controls u; the state d->y is set by the caller
360  const casadi_ivp_prob<T1>* p = d->prob;
361  const casadi_ivp_prob<T1>* pn;
362  d->t = t;
363  d->u = u;
364  d->reset = 1;
365  d->e_valid = 0;
366  if (first_call) {
367  d->n_events = 0;
368  d->ev_restart = 0;
369  d->tape_pos = 0;
370  casadi_ivp_clear_stats(d);
371  // Every evaluation starts from the same controller state
372  d->h_next = 0;
373  d->err_prev = 1;
374  d->last_rejected = 0;
375  }
376  if (d->nom) {
377  // Initial nominal states: the leading entries of x, z and q; afterwards, the nominal
378  // integration continues its own trajectory
379  pn = d->nom->prob;
380  if (first_call) {
381  casadi_copy(d->y, pn->nx, d->nom->y);
382  casadi_copy(d->y + p->nx, pn->nz, d->nom->y + pn->nx);
383  casadi_copy(d->y + p->nx + p->nz, pn->nq, d->nom->y + pn->nx + pn->nz);
384  }
385  casadi_ivp_reset(d->nom, t, u, first_call);
386  }
387 }

390 template<typename T1>
391 casadi_int casadi_ivp_steps(T1 len, T1 step_size) {
392  // Number of equal steps covering len, of at most step_size up to rounding
393  casadi_int n = static_cast<casadi_int>(ceil(len / step_size - 1e-9));
394  return n < 1 ? 1 : n;

397 // SYMBOL "ivp_begin_run"
398 template<typename T1>
399 void casadi_ivp_begin_run(casadi_ivp_data<T1>* d) {
400  // Handle a pending reset: with fixed steps, a new run of steps up to the stop time
401  const casadi_ivp_prob<T1>* p = d->prob;
402  if (p->step_size > 0) {
403  d->run_t0 = d->t;
404  d->run_t1 = d->t_stop;
405  d->run_n = casadi_ivp_steps(d->t_stop - d->t, p->step_size);
406  d->run_h = (d->run_t1 - d->run_t0) / static_cast<T1>(d->run_n);
407  d->run_k = 0;
408  }
409  d->reset = 0;
410  d->jump = 0;
411 }

414 template<typename T1>
415 void casadi_ivp_advance(casadi_ivp_data<T1>* d, T1 t_out, T1 t_stop, int taping) {
416  // Output at t_out, stepping no further than t_stop; tasks follow, see the method

536 // SYMBOL "ivp_tape_full"
537 template<typename T1>
538 int casadi_ivp_tape_full(casadi_ivp_data<T1>* d, casadi_ivp_record_t record) {
539  // No room for a record of this type (a step with events: and an event record)? Sets the
540  // capacity needed
541  const casadi_ivp_prob<T1>* p = d->prob;
542  casadi_int n = (record == IVP_RECORD_STEP ? p->ny * p->n_stages : 0) + p->ny + p->nu + 5;
543  if (record == IVP_RECORD_STEP && p->ne > 0) n += p->ny + p->nu + 5;
544  d->tape_need = d->tape_pos + n;
545  return n > d->tape_cap - d->tape_pos;
546 }

553  if (d->prob->tape_size > 0) return casadi_ivp_stop(d, IVP_TAPE_FAILED);
554  return casadi_ivp_request(d, IVP_TAPE_RESERVE, next);

557 // SYMBOL "ivp_tape_push"
558 template<typename T1>
559 void casadi_ivp_tape_push(casadi_ivp_data<T1>* d, const T1* y0, T1 h, T1 t0, T1 t1) {
560  // Append a record of type d->record from t0 to t1, for which room has been reserved
561  const casadi_ivp_prob<T1>* p = d->prob;
562  casadi_int n_u = d->record == IVP_RECORD_STEP ? p->ny * p->n_stages : 0;
563  T1* rec = d->tape + d->tape_pos;
564  casadi_copy(d->U, n_u, rec);
565  rec += n_u;
566  casadi_copy(y0, p->ny, rec);
567  casadi_copy(d->u_step, p->nu, rec + p->ny);
568  rec += p->ny + p->nu;
569  rec[0] = h;
570  rec[1] = t0;
571  rec[2] = t1;
572  rec[3] = d->t_out;
573  rec[4] = d->record;
574  d->tape_pos += n_u + p->ny + p->nu + 5;
575 }

577 // SYMBOL "ivp_tape_pop"
578 template<typename T1>
579 int casadi_ivp_tape_pop(casadi_ivp_data<T1>* d) {
580  // Restore the last record, clear the seeds of its stages; returns 1 if the tape is empty
581  const casadi_ivp_prob<T1>* p = d->prob;
582  casadi_int n_u;
583  T1* rec;
584  if (d->tape_pos < p->ny + p->nu + 5) return 1;
585  rec = d->tape + d->tape_pos;
586  d->record = static_cast<casadi_ivp_record_t>(static_cast<int>(rec[-1]));
587  d->t_record = rec[-2];
588  d->t1 = rec[-3];
589  d->t0 = rec[-4];
590  d->h = rec[-5];
591  n_u = d->record == IVP_RECORD_STEP ? p->ny * p->n_stages : 0;
592  rec -= p->ny + p->nu + 5;
593  casadi_copy(rec, p->ny, d->y0);
594  casadi_copy(rec + p->ny, p->nu, d->u_step);
595  casadi_copy(rec - n_u, n_u, d->U);
596  d->tape_pos -= n_u + p->ny + p->nu + 5;
597  casadi_clear(d->rU, p->nr * p->n_stages);
598  casadi_clear(d->ry0, p->nr);
599  casadi_clear(d->ru_step, p->nru);
600  return 0;
601 }

620 void casadi_ivp_dense_weights(casadi_ivp_data<T1>* d, T1 theta) {

698 // SYMBOL "ivp_accept"
699 template<typename T1>
700 void casadi_ivp_accept(casadi_ivp_data<T1>* d) {
701  // The step from t0, with y the solution at its end, is accepted
702  const casadi_ivp_prob<T1>* p = d->prob;

713  if (d->taping) {

720 // SYMBOL "ivp_back_begin"
721 template<typename T1>
722 int casadi_ivp_back_begin(casadi_ivp_data<T1>* d) {
723  // Record to process next; returns 0 when the retreat is done (tape exhausted, or the
724  // record belongs to an earlier output interval and becomes pending). An event belongs to the
725  // output interval that contains it
726  T1 r;
727  if (d->status != IVP_SUCCESS) return casadi_ivp_stop(d, d->status);
728  for (;;) {
729  if (!d->record_pending && casadi_ivp_tape_pop(d)) return casadi_ivp_stop(d, IVP_SUCCESS);
730  r = d->record == IVP_RECORD_EVENT ? d->t0 : d->t_record;
731  // Records of later output intervals are skipped with zero adjoint
732  if (r > d->t_start) {
733  d->record_pending = 0;
734  continue;
735  }
736  if (d->record == IVP_RECORD_EVENT ? r <= d->t_out : r < d->t_start) {
737  d->record_pending = 1;
738  d->t = d->t_out;
739  return casadi_ivp_stop(d, IVP_SUCCESS);
740  }
741  d->record_pending = 0;
742  return 1;
743  }
744 }

747 template<typename T1>
748 void casadi_ivp_back_step_begin(casadi_ivp_data<T1>* d) {
749  // Stage adjoints from y1 = y0 + sum_i m_i U_i, added to the seeds of dense outputs
750  const casadi_ivp_prob<T1>* p = d->prob;

754  casadi_axpy(p->nru, 1., d->ru_step, d->adj_u);

758 // SYMBOL "ivp_impulse_begin"
759 template<typename T1>
760 int casadi_ivp_impulse_begin(casadi_ivp_data<T1>* d) {
761  // Pops the record containing t_seed (records of later intervals had zero seeds); returns
762  // 1 if t_seed is inside its step, with the dense output at t_seed in y_out
763  const casadi_ivp_prob<T1>* p = d->prob;
764  while (!d->record_pending && !casadi_ivp_tape_pop(d)) {
765  d->record_pending = (d->record == IVP_RECORD_EVENT ? d->t0 : d->t_record) <= d->t_seed;
766  }

768  // Seed of the state at the end of the pending record
769  casadi_axpy(p->nr, 1., d->seed, d->ry);
770  return casadi_ivp_stop(d, IVP_SUCCESS);
771  }
772  casadi_ivp_interpolate(d, d->t_seed, d->y_out);
773  return 1;
