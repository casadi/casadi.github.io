64  void Erk::init(const Dict& opts) {
65  // Call the base class init
66  RuntimeIntegrator::init(opts);
67 
68  // Default options
69  method_ = "rk45";
70 
71  // Read options
72  for (auto&& op : opts) {
73  if (op.first == "method") {
74  method_ = op.second.to_string();
75  }
76  }
77  casadi_assert(nz_ == 0, "'erk' does not support algebraic states, use 'rosenbrock'");
78 
79  // Reverse communication problem
80  set_coefficients();
81  if (m_err_.empty() && step_size_ == 0) step_size_ = (tout_.back() - t0_) / 20;
82  set_ivp_prob();
83  set_nominal_prob();
84  set_erk_prob();
85 
86  // Work vectors of the kernel
87  alloc_ivp_work();
88  }

90  void Erk::set_coefficients() {
91  if (method_ == "rk4") {
92  // Classical method of order 4
93  a_ = {0, 0, 0, 0,
94  0.5, 0, 0, 0,
95  0, 0.5, 0, 0,
96  0, 0, 1, 0};
97  alpha_ = {0, 0.5, 0.5, 1};
98  m_ = {1./6, 1./3, 1./3, 1./6};
99  // Dense output of order 3: m_i(theta), coefficients of theta, theta^2, theta^3
100  n_dense_ = 3;
101  m_dense_ = {1, 0, 0, 0,
102  -1.5, 1, 1, -0.5,
103  2./3, -2./3, -2./3, 2./3};
104  // No error estimate
105  m_err_.clear();
106  order_err_ = 0;
107  } else if (method_ == "rk45") {
108  // Pair of order 5(4), first same as last, derived by a standalone tool
109  a_ = {0, 0, 0, 0, 0, 0, 0,
110  2./9, 0, 0, 0, 0, 0, 0,
111  95./1024, 225./1024, 0, 0, 0, 0, 0,
112  595749./800000, -558657./160000, 44649./12500, 0, 0, 0, 0,
113  8133487./3056130, -547./42, 22178368./1898505, -4601000./12530133, 0, 0, 0,
114  418551./146000, -404253./29200, 197614472./16087375, -74305./320251,
115  -4991679./67174600, 0, 0,
116  629./6600, 0, 677888./1454475, 208000./337799, -453789./920200, 73./231, 0};
117  alpha_ = {0, 2./9, 5./16, 33./40, 20./21, 1, 1};
118  m_ = {629./6600, 0, 677888./1454475, 208000./337799, -453789./920200, 73./231, 0};
119  // Dense output of order 4, continuously differentiable: coefficients of theta, ..., theta^4
120  n_dense_ = 4;
121  m_dense_ = {1, 0, 0, 0, 0, 0, 0,
122  -3063./1100, 0, 1948928./484825, -1404000./337799, 2398599./460100,
123  -292./77, 3./2,
124  2434./825, 0, -8982016./1454475, 520000./48257, -1426194./115025,
125  292./33, -4,
126  -471./440, 0, 254208./96965, -2028000./337799, 1231713./184040,
127  -365./77, 5./2};
128  // Error estimate: difference with the embedded solution of order 4
129  m_err_ = {89834./99443025, 0, -183980032./58439351025, 2874688000./122151834189,
130  -92439186./1540529825, 468368./9281349, -8./681};
131  order_err_ = 4;
132  } else {
133  casadi_error("'erk': unknown method '" + method_ + "', expected 'rk45' or 'rk4'");
134  }
135  }

137  void Erk::set_erk_prob() {
138  // First same as last: the last stage is evaluated at t0 + h and y1
139  casadi_int s = alpha_.size();
140  pe_.fsal = alpha_[s - 1] == 1;
141  for (casadi_int j = 0; j < s; ++j) pe_.fsal = pe_.fsal && a_[(s - 1) * s + j] == m_[j];
142  }
