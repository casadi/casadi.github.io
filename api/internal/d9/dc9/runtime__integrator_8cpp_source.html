77 void RuntimeIntegrator::init(const Dict& opts) {
78  // Control groups, before the base class sizes the controls; a single one without controls
79  group_start_ = range(nt() + 1);
80  if (oracle_.numel_in(DYN_U) == 0) group_start_ = {0, nt()};
81  auto cgs = opts.find("control_group_size");
82  if (cgs != opts.end()) {
83  group_start_ = {0};
84  for (casadi_int n : cgs->second.to_int_vector()) {
85  casadi_assert(n > 0, "'control_group_size' entries must be positive");
86  group_start_.push_back(group_start_.back() + n);
87  }
88  casadi_assert(group_start_.back() == nt(), "'control_group_size' must sum to the number "
89  "of output times, " + str(nt()) + ", got " + str(group_start_.back()));
90  }
91 
92  // Call the base class init
93  Integrator::init(opts);
94 
95  // Controls and control adjoints by output interval, for sparsity propagation
96  if (ngroup() < nt()) alloc_w((nu_ + nuq_) * nt(), true);
97 
98  // Default options
99  step_size_ = 0;
100  abstol_ = 1e-8;
101  reltol_ = 1e-6;
102  max_num_steps_ = 10000;
103  step0_ = 0;
104  max_step_size_ = 0;
105  min_step_size_ = 1e-14;
106  quad_err_con_ = true;
107  max_tape_steps_ = 0;
108 
109  // Read options
110  for (auto&& op : opts) {
111  if (op.first == "step_size") {
112  step_size_ = op.second;
113  } else if (op.first == "abstol") {
114  abstol_ = op.second;
115  } else if (op.first == "reltol") {
116  reltol_ = op.second;
117  } else if (op.first == "max_num_steps") {
118  max_num_steps_ = op.second;
119  } else if (op.first == "step0") {
120  step0_ = op.second;
121  } else if (op.first == "max_step_size") {
122  max_step_size_ = op.second;
123  } else if (op.first == "min_step_size") {
124  min_step_size_ = op.second;
125  } else if (op.first == "quad_err_con") {
126  quad_err_con_ = op.second;
127  } else if (op.first == "max_tape_steps") {
128  max_tape_steps_ = op.second;
129  }
130  }
131  casadi_assert(step_size_ >= 0, "'step_size' must be nonnegative");
132 
133  // Error control on the nominal states only: derivative integrators share the mesh
134  if (derivative_of_.is_null()) {
135  nx_err_ = nx1_;
136  nz_err_ = nz1_;
137  nq_err_ = nq1_;
138  } else {
139  const RuntimeIntegrator* d = derivative_of_.get<RuntimeIntegrator>();
140  casadi_assert_dev(d != nullptr);
141  nx_err_ = d->nx_err_;
142  nz_err_ = d->nz_err_;
143  nq_err_ = d->nq_err_;
144  }
145 
146  // Right-hand side [ode; alg; quad] and its vector-Jacobian product, one function each
147  create_function("rhsF", dyn_in(), {"ode", "alg", "quad"});
148  if (nadj_ > 0) {
149  create_function(rdae_, "vjpB", bdyn_in(), {"adj_x", "adj_z", "adj_p", "adj_u"});
150  }
151 
152  // Zero-crossing functions and their time derivative; the jump at an event, its derivatives
153  if (ne_ > 0) {
154  create_function("zeroF", dyn_in(), {"zero"});
155  create_forward("zeroF", 1);
156  if (has_function("transition")) {
157  Function jump = jump_function();
158  set_function(jump, "jumpF");
159  if (nfwd_ > 0) create_forward("jumpF", nfwd_);
160  if (nadj_ > 0) {
161  // Adjoint seeds of x, z, q to adjoints of x, z, p, u
162  std::vector<MX> in = jump.mx_in(), out = jump(in), seed;
163  for (casadi_int i = 0; i < jump.n_out(); ++i) {
164  Sparsity sp = repmat(jump.sparsity_out(i), 1, nadj_);
165  seed.push_back(MX::sym("adj_" + jump.name_out(i), sp));
166  }
167  std::vector<MX> arg = in;
168  arg.insert(arg.end(), out.begin(), out.end());
169  arg.insert(arg.end(), seed.begin(), seed.end());
170  std::vector<MX> adj = jump.reverse(nadj_)(arg);
171  in.insert(in.end(), seed.begin(), seed.end());
172  set_function(Function("jumpB", in, {adj[1], adj[2], adj[4], adj[5]}), "jumpB");
173  if (nfwd_ > 0) create_forward("jumpB", nfwd_);
174  }
175  }
176  }
177 
178  // Jacobian blocks w.r.t. x, z and time, in the order of casadi_ivp_jac_t
179  sp_jac_.clear();
180  if (needs_jacobian()) {
181  Function jac = create_function("jacF", {"t", "x", "z", "p", "u"},
182  {"jac:ode:x", "jac:alg:x", "jac:quad:x", "jac:ode:z", "jac:alg:z", "jac:quad:z",
183  "jac:ode:t", "jac:alg:t", "jac:quad:t"});
184  for (casadi_int k = 0; k < IVP_NUM_JAC; ++k) sp_jac_.push_back(jac.sparsity_out(k));
185  if (nadj_ > 0) set_function(jac.reverse(nadj_), "jacB");
186  }
187 
188  // Forward sensitivities of all of these
189  if (nfwd_ > 0) {
190  create_forward("rhsF", nfwd_);
191  if (nadj_ > 0) create_forward("vjpB", nfwd_);
192  if (needs_jacobian()) {
193  create_forward("jacF", nfwd_);
194  if (nadj_ > 0) create_forward("jacB", nfwd_);
195  }
196  }

199 Function RuntimeIntegrator::jump_function() const {
200  // State after the event selected by w as a function of the state before it: moved along
201  // the trajectory to the event surface, through the transition and back to t, to second
202  // order in the time s to the surface. At an event s vanishes, so the value is that of the
203  // transition and the first and second derivatives are those of the event, including the
204  // dependence of its time. Algebraic states are consistent along the way: the given ones
205  // are only starting points
206  MX t = MX::sym("t"), s = MX::sym("s"), w = MX::sym("w", ne_), q = MX::sym("q", nq1_);
207  MX x = MX::sym("x", oracle_.sparsity_in(DYN_X)), z = MX::sym("z", oracle_.sparsity_in(DYN_Z));
208  MX p = MX::sym("p", oracle_.sparsity_in(DYN_P)), u = MX::sym("u", oracle_.sparsity_in(DYN_U));
209  const Function& tr = get_function("transition");
210  // Argument i of f, empty if f does not depend on it
211  auto arg = [](const Function& f, casadi_int i, const MX& v) -> MX {
212  return f.nnz_in(i) == 0 ? MX(f.sparsity_in(i)) : v;
213  };
214  auto dae = [&](const MX& tt, const MX& xx, const MX& zz) -> std::vector<MX> {
215  return oracle_(std::vector<MX>{arg(oracle_, DYN_T, tt), xx, zz, p, u});
216  };
217  // Algebraic states consistent with (tt, xx), from zz
218  Function z_solve;
219  if (nz1_ > 0) {
220  MX zs = MX::sym("z", z.sparsity()), ts = MX::sym("t"), xs = MX::sym("x", x.sparsity());
221  Function alg("alg", {zs, ts, xs, p, u}, {dae(ts, xs, zs)[DYN_ALG]});
222  z_solve = rootfinder("z_consistent", "fast_newton", alg,
223  Dict{{"abstol", 1e-14}, {"abstolStep", 1e-14}, {"max_iter", 50}});
224  }
225  auto consistent = [&](const MX& tt, const MX& xx, const MX& zz) -> MX {
226  return nz1_ == 0 ? zz : z_solve(std::vector<MX>{zz, tt, xx, p, u}).at(0);
227  };
228  // Rates of x, q at (tt, xx) and their derivatives along the trajectory
229  auto rates = [&](const MX& tt, const MX& xx, const MX& zz, std::vector<MX>& r) -> void {
230  std::vector<MX> r0 = dae(tt, xx, consistent(tt, xx, zz));
231  MX x1 = xx + s * r0[DYN_ODE];
232  std::vector<MX> r1 = dae(tt + s, x1, consistent(tt + s, x1, zz));
233  r = MX::substitute({r0[DYN_ODE], r0[DYN_QUAD], MX::jacobian(r1[DYN_ODE], s),
234  MX::jacobian(r1[DYN_QUAD], s)}, {s}, {MX(0)});
235  };
236  // Time to the surface of the selected zero-crossing function
237  std::vector<MX> r;
238  rates(t, x, z, r);
239  MX x_s = x + s * r[0] + 0.5 * s * s * r[2];
240  MX g = MX::dot(w, dae(t + s, x_s, consistent(t + s, x_s, z))[DYN_ZERO]);
241  MX g_s = MX::jacobian(g, s);
242  std::vector<MX> gv = MX::substitute({g, g_s, MX::jacobian(g_s, s)}, {s}, {MX(0)});
243  MX s1 = -gv[0] / gv[1];
244  MX ss = s1 - 0.5 * gv[2] * s1 * s1 / gv[1];
245  // Transition on the surface
246  std::vector<double> index(ne_);
247  for (casadi_int i = 0; i < ne_; ++i) index[i] = static_cast<double>(i);
248  MX x_ev = x + ss * r[0] + 0.5 * ss * ss * r[2];
249  MX q_ev = q + ss * r[1] + 0.5 * ss * ss * r[3];
250  std::vector<MX> post = tr(std::vector<MX>{arg(tr, EVENT_INDEX, MX::dot(w, MX(DM(index)))),
251  arg(tr, EVENT_T, t + ss), x_ev, consistent(t + ss, x_ev, z), arg(tr, EVENT_P, p),
252  arg(tr, EVENT_U, u)});
253  // Back to t along the trajectory after the event
254  MX z_post = consistent(t + ss, post[EVENT_POST_X], post[EVENT_POST_Z]);
255  rates(t + ss, post[EVENT_POST_X], z_post, r);
256  MX x_post = post[EVENT_POST_X] - ss * r[0] + 0.5 * ss * ss * r[2];
257  MX q_post = q_ev - ss * r[1] + 0.5 * ss * ss * r[3];
258  return Function("jumpF", {t, x, z, q, p, u, w}, {x_post, z_post, q_post},
259  {"t", "x", "z", "q", "p", "u", "w"}, {"post_x", "post_z", "post_q"});
260 }

313 void RuntimeIntegrator::set_nominal_prob() {
314  // Forward directions folded into the DAE change the arithmetic of the nominal states: with
315  // adaptive steps, replay the steps of a nominal integration with the original functions
316  const RuntimeIntegrator* r = root();
317  if (!has_function("rhsF_nom") && r != this && r->nx1_ != nx1_ && step_size_ == 0) {
318  for (auto&& calls : r->task_calls("")) {
319  for (auto&& c : calls) {
320  std::string fname = c.fname + "_nom";
321  if (!has_function(fname)) set_function(r->get_function(c.fname), fname);
322  }
323  }
324  }
325  replay_ = has_function("rhsF_nom");
326  if (!replay_) return;
327  calls_nom_ = root()->task_calls("_nom");
328  p_nom_ = root()->p_;
329  p_nom_.tape_size = 0;
330  casadi_ivp_setup(&p_nom_);
331 }

343 casadi_int RuntimeIntegrator::tape_capacity() const {
344  // Steps: a user bound with adaptive steps, a run of steps per control group with fixed steps
345  casadi_int ny = nx_ + nz_ + nq_, n_step = max_tape_steps_, n_init, n_event = 0;
346  if (step_size_ > 0) {
347  n_step = 0;
348  for (casadi_int j = 0; j < ngroup(); ++j) {
349  double t = j == 0 ? t0_ : tout_[group_start_[j] - 1];
350  n_step += casadi_ivp_steps(tout_[group_start_[j + 1] - 1] - t, step_size_);
351  }
352  }
353  if (nadj_ == 0 || n_step == 0) return 0;
354  // Every output interval may start with a consistent initialization
355  n_init = nz_ > 0 ? nt() : 0;
356  if (ne_ > 0) {
357  // Event records, room for one more with each step; after an event, a consistent
358  // initialization and, with fixed steps, a new run of steps
359  n_event = max_events_ + 1;
360  if (nz_ > 0) n_init += max_events_;
361  if (step_size_ > 0) n_step += max_events_;
362  }
363  return n_step * (ny * (static_cast<casadi_int>(alpha_.size()) + 1) + nu_ + 5)
364  + (n_init + n_event) * (ny + nu_ + 5);
365 }

405 int RuntimeIntegrator::advance_noevent(IntegratorMemory* mem) const {
406  auto m = static_cast<RuntimeIntegratorMemory*>(mem);
407  casadi_ivp_data<double>* d = &m->d;
408  casadi_ivp_advance(d, m->t_next, m->t_stop, nrx_ > 0);
409  run(m);
410  casadi_assert(d->status != IVP_TAPE_FAILED, name_ + ": the tape of " + str(max_tape_steps_)
411  + " steps is full at t = " + str(d->t) + ", increase max_tape_steps");
412  casadi_assert(d->status != IVP_TOO_MANY_EVENTS, name_ + ": more than max_events = "
413  + str(max_events_) + " events, the last at t = " + str(d->t_ev));
414  casadi_assert(d->status != IVP_EVENT_FAILED, name_ + ": event near t = " + str(d->t_ev)
415  + " not located within event_acceptable_tol");
416  if (d->status != IVP_SUCCESS) {
417  if (verbose_) {
418  casadi_warning(name_ + ": integration failed at t = " + str(d->t) + ", status "
419  + str(static_cast<casadi_int>(d->status)));
420  }
421  return 1;
422  }
423  casadi_copy(d->y_out, nx_, m->x);
424  casadi_copy(d->y_out + nx_, nz_, m->z);
425  casadi_copy(d->y_out + nx_ + nz_, nq_, m->q);
426  return 0;
427 }

457 int RuntimeIntegrator::eval(const double** arg, double** res,
458  casadi_int* iw, double* w, void* mem) const {
459  auto m = static_cast<RuntimeIntegratorMemory*>(mem);
460  const double* u = arg[INTEGRATOR_U];
461  const double* adj_xf = arg[INTEGRATOR_ADJ_XF];
462  const double* adj_zf = arg[INTEGRATOR_ADJ_ZF];
463  const double* adj_qf = arg[INTEGRATOR_ADJ_QF];
464  double* xf = res[INTEGRATOR_XF];
465  double* zf = res[INTEGRATOR_ZF];
466  double* qf = res[INTEGRATOR_QF];
467  double* adj_u = res[INTEGRATOR_ADJ_U];
468  setup(m, arg + INTEGRATOR_NUM_IN, res + INTEGRATOR_NUM_OUT, iw, w);
469  set_q(m, nullptr);
470  set_x(m, arg[INTEGRATOR_X0]);
471  set_z(m, arg[INTEGRATOR_Z0]);
472  set_p(m, arg[INTEGRATOR_P]);
473  m->num_events = 0;
474  m->t = t0_;
475  // Forward integration: a run of steps per control group
476  for (casadi_int j = 0; j < ngroup(); ++j) {
477  set_u(m, u ? u + j * nu_ : nullptr);
478  m->t_stop = tout_[group_start_[j + 1] - 1];
479  reset(m, j == 0);
480  for (m->k = group_start_[j]; m->k < group_start_[j + 1]; ++m->k) {
481  m->t_next = tout_[m->k];
482  if (advance_noevent(m)) return 1;
483  m->t = m->t_next;
484  get_x(m, xf ? xf + m->k * nx_ : nullptr);
485  get_z(m, zf ? zf + m->k * nz_ : nullptr);
486  get_q(m, qf ? qf + m->k * nq_ : nullptr);
487  }
488  }
489  // Discrete adjoint, output interval by output interval
490  if (nrx_ > 0) {
491  resetB(m);
492  for (casadi_int j = ngroup(); j-- > 0; ) {
493  // Control adjoints of this group only
494  casadi_clear(m->d.adj_u, nuq_);
495  for (m->k = group_start_[j + 1]; m->k-- > group_start_[j]; ) {
496  m->t = tout_[m->k];
497  impulseB(m, adj_xf ? adj_xf + m->k * nrx_ : nullptr,
498  adj_zf ? adj_zf + m->k * nrz_ : nullptr, adj_qf ? adj_qf + m->k * nrp_ : nullptr);
499  m->t_next = m->k > 0 ? tout_[m->k - 1] : t0_;
500  retreat(m, nullptr, m->k == 0 ? res[INTEGRATOR_ADJ_X0] : nullptr,
501  m->k == 0 ? res[INTEGRATOR_ADJ_P] : nullptr,
502  adj_u && m->k == group_start_[j] ? adj_u + j * nuq_ : nullptr);
503  }
504  }
505  }
506  join_results(m);
507  if (print_stats_) print_stats(m);
508  return 0;
509 }

512  // Dependencies with a column per control group to one with a column per output interval
513  void groups_to_intervals(const std::vector<casadi_int>& gs, const bvec_t* g, bvec_t* k,

518  }

520  void intervals_to_groups(const std::vector<casadi_int>& gs, const bvec_t* k, bvec_t* g,
521  casadi_int n) {
522  for (size_t j = 0; j + 1 < gs.size(); ++j) {
523  for (casadi_int i = gs[j]; i < gs[j + 1]; ++i) {
524  for (casadi_int l = 0; l < n; ++l) g[j * n + l] |= k[i * n + l];
525  }
526  }

530 int RuntimeIntegrator::sp_forward(const bvec_t** arg, bvec_t** res,
531  casadi_int* iw, bvec_t* w, void* mem) const {
532  if (ne_ > 0 || ngroup() == nt()) return Integrator::sp_forward(arg, res, iw, w, mem);
533  // Controls and control adjoints by output interval
534  const bvec_t* u = arg[INTEGRATOR_U];
535  bvec_t* adj_u = res[INTEGRATOR_ADJ_U];
536  bvec_t* u_k = w; w += nu_ * nt();
537  bvec_t* adj_u_k = w; w += nuq_ * nt();
538  if (u) groups_to_intervals(group_start_, u, u_k, nu_);
539  arg[INTEGRATOR_U] = u ? u_k : nullptr;
540  res[INTEGRATOR_ADJ_U] = adj_u ? adj_u_k : nullptr;
541  int flag = Integrator::sp_forward(arg, res, iw, w, mem);
542  arg[INTEGRATOR_U] = u;
543  res[INTEGRATOR_ADJ_U] = adj_u;
544  if (adj_u) {
545  std::fill_n(adj_u, nuq_ * ngroup(), 0);
546  intervals_to_groups(group_start_, adj_u_k, adj_u, nuq_);
547  }
548  return flag;
549 }

551 int RuntimeIntegrator::sp_reverse(bvec_t** arg, bvec_t** res,
552  casadi_int* iw, bvec_t* w, void* mem) const {
553  if (ne_ > 0 || ngroup() == nt()) return Integrator::sp_reverse(arg, res, iw, w, mem);
554  // Controls and control adjoints by output interval
555  bvec_t* u = arg[INTEGRATOR_U];
556  bvec_t* adj_u = res[INTEGRATOR_ADJ_U];
557  bvec_t* u_k = w; w += nu_ * nt();
558  bvec_t* adj_u_k = w; w += nuq_ * nt();

566  int flag = Integrator::sp_reverse(arg, res, iw, w, mem);
567  arg[INTEGRATOR_U] = u;
568  res[INTEGRATOR_ADJ_U] = adj_u;
569  if (u) intervals_to_groups(group_start_, u_k, u, nu_);

587 int RuntimeIntegrator::task(RuntimeIntegratorMemory* m) const {
588  casadi_ivp_data<double>* d = &m->d;
589  if (d->task == IVP_TAPE_RESERVE) {
590  // The tape persists between evaluations: grow it only when too small
591  if (m->tape.size() < static_cast<size_t>(d->tape_need)) {
592  m->tape.resize(std::max(static_cast<size_t>(d->tape_need), 2 * m->tape.size()));
593  }
594  d->tape = get_ptr(m->tape);
595  d->tape_cap = m->tape.size();
596  return 0;
597  }
598  return eval_task(m, d, calls_);

601 int RuntimeIntegrator::eval_task(RuntimeIntegratorMemory* m, casadi_ivp_data<double>* d,
602  const std::vector<std::vector<TaskCall>>& calls) const {
603  // Adaptive step size control, or function evaluations as described by the task table
604  if (d->task == IVP_STEP_CONTROL) {
605  casadi_ivp_control(d);
606  return 0;
607  }
608  if (d->task == IVP_EVENT) {
609  while (casadi_ivp_event(d)) {
610  if (eval_task(m, d, calls)) d->fail = 1;
611  }
612  return 0;
613  }
614  if (d->task == IVP_EVENT_ADJ) {
615  while (casadi_ivp_event_adj(d)) {
616  if (eval_task(m, d, calls)) d->fail = 1;
617  }
618  return 0;
619  }
620  for (const TaskCall& c : calls.at(d->task)) {
621  std::fill_n(m->arg, c.n_in, nullptr);
622  std::fill_n(m->res, c.n_out, nullptr);
623  for (const Slot& e : c.arg) m->arg[e.i] = slot_ptr(m, d, e);
624  for (const Slot& e : c.res) m->res[e.i] = slot_ptr(m, d, e);
625  if (calc_function(m, c.fname)) return 1;
626  }
627  return 0;
628 }

630 double* RuntimeIntegrator::slot_ptr(RuntimeIntegratorMemory* m, casadi_ivp_data<double>* d,

637  case SRC_P: b = m->p; break;
638  case SRC_U: b = d->u_step; break;
639  case SRC_RES: b = d->res; break;
640  case SRC_W: b = const_cast<double*>(d->w); break;
641  case SRC_JAC: b = d->jac; break;
642  case SRC_JAC_FWD: b = d->jac_fwd; break;
643  case SRC_JAC_ADJ: b = d->jac_adj; break;
644  case SRC_JAC_ADJ_FWD: b = d->jac_adj_fwd; break;

646  case SRC_VJP_Z: b = d->vjp_z; break;
647  case SRC_VJP_P: b = d->vjp_p; break;
648  case SRC_VJP_U: b = d->vjp_u; break;
649  case SRC_EV_ONE: b = &d->ev_one; break;
650  case SRC_EV_W: b = d->ev_w; break;
651  case SRC_YDOT: b = d->ydot; break;
652  case SRC_EDOT: b = d->edot; break;
653  }
654  return b + e.off;

683 std::vector<std::vector<RuntimeIntegrator::TaskCall>> RuntimeIntegrator::task_calls(
684  const std::string& suffix) const {
685  // For each task, the functions to call and the buffers of their arguments and results
686  std::vector<std::vector<TaskCall>> calls(IVP_TAPE_RESERVE);
687  static_assert(static_cast<int>(BDYN_U) == static_cast<int>(DYN_U)
688  && static_cast<int>(BDYN_T) == static_cast<int>(DYN_T),
689  "Backward DAE inputs start with t, x, z, p, u");
690  const casadi_int nj = IVP_NUM_JAC;
691  // Results of the adjoint functions: adjoints of x, z, p, u and their forward derivatives
692  const Src vjp[4] = {SRC_VJP_X, SRC_VJP_Z, SRC_VJP_P, SRC_VJP_U};
693  const casadi_int n_vjp[4] = {nrx1_ * nadj_, nrz1_ * nadj_, nrq1_ * nadj_, nuq1_ * nadj_};
694  auto call = [&](const std::string& fname) -> TaskCall {
695  const Function& f = get_function(fname);
696  return {fname + suffix, f.n_in(), f.n_out(), {}, {}};
697  };
698  // Inputs t, x, z, p, u at offset i; forward seeds are the sensitivity parts
699  auto dyn = [&](TaskCall& c, casadi_int i, bool fwd) -> void {
700  if (!fwd) c.arg.push_back({i + DYN_T, SRC_T, 0});
701  c.arg.push_back({i + DYN_X, SRC_Y, fwd ? nx1_ : 0});
702  c.arg.push_back({i + DYN_Z, SRC_Y, nx_ + (fwd ? nz1_ : 0)});
703  c.arg.push_back({i + DYN_P, SRC_P, fwd ? np1_ : 0});
704  c.arg.push_back({i + DYN_U, SRC_U, fwd ? nu1_ : 0});
705  };
706  // Jacobian blocks in buffer src with ndir directions, block k at slot i + k
707  auto blocks = [&](std::vector<Slot>& v, casadi_int i, Src src, casadi_int ndir) -> void {
708  for (casadi_int k = 0; k < nj; ++k) v.push_back({i + k, src, ndir * p_.jac_off[k]});
709  };
710  // Right-hand side [ode; alg; quad] at (t, y)
711  TaskCall rhs = call("rhsF");
712  dyn(rhs, 0, false);
713  rhs.res = {{0, SRC_RES, 0}, {1, SRC_RES, nx_}, {2, SRC_RES, nx_ + nz_}};
714  calls[IVP_EVAL_RHS].push_back(rhs);
715  if (nfwd_ > 0) {
716  TaskCall c = call(forward_name("rhsF", nfwd_));
717  dyn(c, 0, false);
718  for (casadi_int k = 0; k < 3; ++k) {
719  c.arg.push_back({DYN_NUM_IN + k, SRC_RES, rhs.res[k].off});
720  }
721  dyn(c, DYN_NUM_IN + 3, true);
722  c.res = {{0, SRC_RES, nx1_}, {1, SRC_RES, nx_ + nz1_}, {2, SRC_RES, nx_ + nz_ + nq1_}};
723  calls[IVP_EVAL_RHS].push_back(c);
724  }
725  if (needs_jacobian()) {
726  // Jacobian blocks; forward derivatives give the augmented off-diagonal blocks
727  TaskCall jac = call("jacF");
728  dyn(jac, 0, false);
729  blocks(jac.res, 0, SRC_JAC, 1);
730  calls[IVP_EVAL_JAC].push_back(jac);
731  if (nfwd_ > 0) {
732  TaskCall c = call(forward_name("jacF", nfwd_));
733  dyn(c, 0, false);
734  blocks(c.arg, DYN_NUM_IN, SRC_JAC, 1);
735  dyn(c, DYN_NUM_IN + nj, true);
736  blocks(c.res, 0, SRC_JAC_FWD, nfwd_);
737  calls[IVP_EVAL_JAC].push_back(c);
738  }
739  }
740  if (ne_ > 0) {
741  // Zero-crossing functions; derivative in the direction (1, ydot); forward sensitivities
742  TaskCall zero = call("zeroF");
743  dyn(zero, 0, false);
744  zero.res = {{0, SRC_RES, 0}};
745  calls[IVP_EVAL_ZERO].push_back(zero);
746  TaskCall c = call(forward_name("zeroF", 1));
747  dyn(c, 0, false);
748  c.arg.push_back({DYN_NUM_IN, SRC_RES, 0});
749  c.arg.push_back({DYN_NUM_IN + 1 + DYN_T, SRC_EV_ONE, 0});
750  c.arg.push_back({DYN_NUM_IN + 1 + DYN_X, SRC_YDOT, 0});
751  c.arg.push_back({DYN_NUM_IN + 1 + DYN_Z, SRC_YDOT, nx_});
752  c.res = {{0, SRC_EDOT, 0}};
753  calls[IVP_EVAL_ZERO].push_back(c);
754  if (has_function("jumpF")) {
755  // Inputs t, x, z, q, p, u, w of the jump at offset i; forward seeds are the sensitivities
756  auto jump = [&](TaskCall& c, casadi_int i, bool fwd) -> void {
757  if (!fwd) c.arg.push_back({i, SRC_T, 0});
758  c.arg.push_back({i + 1, SRC_Y, fwd ? nx1_ : 0});
759  c.arg.push_back({i + 2, SRC_Y, nx_ + (fwd ? nz1_ : 0)});
760  c.arg.push_back({i + 3, SRC_Y, nx_ + nz_ + (fwd ? nq1_ : 0)});
761  c.arg.push_back({i + 4, SRC_P, fwd ? np1_ : 0});
762  c.arg.push_back({i + 5, SRC_U, fwd ? nu1_ : 0});
763  if (!fwd) c.arg.push_back({i + 6, SRC_EV_W, 0});
764  };
765  TaskCall jf = call("jumpF");
766  jump(jf, 0, false);
767  jf.res = {{0, SRC_RES, 0}, {1, SRC_RES, nx_}, {2, SRC_RES, nx_ + nz_}};
768  calls[IVP_EVAL_JUMP].push_back(jf);
769  if (nfwd_ > 0) {
770  c = call(forward_name("jumpF", nfwd_));
771  jump(c, 0, false);
772  for (casadi_int k = 0; k < 3; ++k) c.arg.push_back({7 + k, SRC_RES, jf.res[k].off});
773  jump(c, 10, true);
774  c.res = {{0, SRC_RES, nx1_}, {1, SRC_RES, nx_ + nz1_}, {2, SRC_RES, nx_ + nz_ + nq1_}};
775  calls[IVP_EVAL_JUMP].push_back(c);
776  }
777  if (nadj_ > 0) {
778  // Adjoint seeds of x, z, q to adjoints of x, z, p, u, with forward-over-adjoint
779  TaskCall jb = call("jumpB");
780  jump(jb, 0, false);
781  jb.arg.push_back({7, SRC_W, 0});
782  jb.arg.push_back({8, SRC_W, nrx_});
783  jb.arg.push_back({9, SRC_W, nrx_ + nrz_});
784  for (casadi_int k = 0; k < 4; ++k) jb.res.push_back({k, vjp[k], 0});
785  calls[IVP_EVAL_JUMP_ADJ].push_back(jb);
786  if (nfwd_ > 0) {
787  c = call(forward_name("jumpB", nfwd_));
788  c.arg = jb.arg;
789  for (casadi_int k = 0; k < 4; ++k) c.arg.push_back({10 + k, vjp[k], 0});
790  jump(c, 14, true);
791  c.arg.push_back({21, SRC_W, nrx1_ * nadj_});
792  c.arg.push_back({22, SRC_W, nrx_ + nrz1_ * nadj_});
793  c.arg.push_back({23, SRC_W, nrx_ + nrz_ + nrp1_ * nadj_});
794  for (casadi_int k = 0; k < 4; ++k) c.res.push_back({k, vjp[k], n_vjp[k]});
795  calls[IVP_EVAL_JUMP_ADJ].push_back(c);
796  }
797  }
798  }
799  }
800  if (nadj_ > 0) {
801  // [dF/dx; dF/dz; dF/dp; dF/du]^T w, with forward-over-adjoint
802  TaskCall vjpb = call("vjpB");
803  dyn(vjpb, 0, false);
804  vjpb.arg.push_back({BDYN_ADJ_ODE, SRC_W, 0});
805  vjpb.arg.push_back({BDYN_ADJ_ALG, SRC_W, nrx_});
806  vjpb.arg.push_back({BDYN_ADJ_QUAD, SRC_W, nrx_ + nrz_});
807  for (casadi_int k = 0; k < 4; ++k) vjpb.res.push_back({k, vjp[k], 0});
808  calls[IVP_EVAL_VJP].push_back(vjpb);
809  if (nfwd_ > 0) {
810  TaskCall c = call(forward_name("vjpB", nfwd_));
811  casadi_int i = BDYN_NUM_IN + 4;
812  c.arg = vjpb.arg;
813  for (casadi_int k = 0; k < 4; ++k) c.arg.push_back({BDYN_NUM_IN + k, vjp[k], 0});
814  dyn(c, i, true);
815  c.arg.push_back({i + BDYN_ADJ_ODE, SRC_W, nrx1_ * nadj_});
816  c.arg.push_back({i + BDYN_ADJ_ALG, SRC_W, nrx_ + nrz1_ * nadj_});
817  c.arg.push_back({i + BDYN_ADJ_QUAD, SRC_W, nrx_ + nrz_ + nrp1_ * nadj_});
818  for (casadi_int k = 0; k < 4; ++k) c.res.push_back({k, vjp[k], n_vjp[k]});
819  calls[IVP_EVAL_VJP].push_back(c);
820  }
821  if (needs_jacobian()) {
822  // d/d(x, z, p, u) <jac_adj, Jacobian blocks>, with forward-over-adjoint
823  TaskCall jacb = call("jacB");
824  dyn(jacb, 0, false);
825  blocks(jacb.arg, DYN_NUM_IN, SRC_JAC, 1);
826  blocks(jacb.arg, DYN_NUM_IN + nj, SRC_JAC_ADJ, nadj_);
827  for (casadi_int k = 0; k < 4; ++k) jacb.res.push_back({DYN_X + k, vjp[k], 0});
828  calls[IVP_EVAL_JAC_ADJ].push_back(jacb);
829  if (nfwd_ > 0) {
830  TaskCall c = call(forward_name("jacB", nfwd_));
831  casadi_int i = DYN_NUM_IN + 2 * nj, i_fwd = i + DYN_NUM_IN;
832  c.arg = jacb.arg;
833  for (casadi_int k = 0; k < 4; ++k) c.arg.push_back({i + DYN_X + k, vjp[k], 0});
834  dyn(c, i_fwd, true);
835  blocks(c.arg, i_fwd + DYN_NUM_IN + nj, SRC_JAC_ADJ_FWD, nadj_ * nfwd_);
836  for (casadi_int k = 0; k < 4; ++k) c.res.push_back({DYN_X + k, vjp[k], n_vjp[k]});
837  calls[IVP_EVAL_JAC_ADJ].push_back(c);
838  }
839  }
840  }
841  return calls;
842 }

1029 void RuntimeIntegrator::codegen_task(CodeGenerator& g, const std::string& d,
1030  const std::vector<std::vector<TaskCall>>& calls, bool in_event) const {
1031  // Straight-line code for the task table, as eval_task: direct calls, no function pointers
1032  static const char* task_name[] = {"IVP_EVAL_RHS", "IVP_EVAL_JAC", "IVP_EVAL_VJP",
1033  "IVP_EVAL_JAC_ADJ", "IVP_EVAL_ZERO", "IVP_EVAL_JUMP", "IVP_EVAL_JUMP_ADJ"};
1034  g << "switch (" << d << ".task) {\n";
1035  for (casadi_int task = 0; task < static_cast<casadi_int>(calls.size()); ++task) {
1036  if (calls[task].empty()) continue;
1037  // Tasks of the events are requested by casadi_ivp_event and casadi_ivp_event_adj only
1038  bool ev = task >= IVP_EVAL_ZERO && task <= IVP_EVAL_JUMP_ADJ;
1039  if (in_event != ev) continue;
1040  g << "case " << task_name[task] << ":\n";
1041  // Slots set earlier in the same case need not be set again
1042  std::map<casadi_int, std::string> arg_set, res_set;
1043  for (const TaskCall& c : calls[task]) {
1044  std::vector<std::string> arg(c.n_in, "0"), res(c.n_out, "0");
1045  for (const Slot& e : c.arg) arg[e.i] = slot_expr(d, e);
1046  for (const Slot& e : c.res) res[e.i] = slot_expr(d, e);
1047  for (casadi_int i = 0; i < c.n_in; ++i) {
1048  if (arg_set.count(i) && arg_set[i] == arg[i]) continue;
1049  g << "sarg[" << i << "] = " << arg[i] << ";\n";
1050  arg_set[i] = arg[i];
1051  }
1052  for (casadi_int i = 0; i < c.n_out; ++i) {
1053  if (res_set.count(i) && res_set[i] == res[i]) continue;
1054  g << "sres[" << i << "] = " << res[i] << ";\n";
1055  res_set[i] = res[i];
1056  }
1057  // On failure, skip the remaining evaluations of the task
1058  std::string flag = g(get_function(c.fname), "sarg", "sres", "iw", "w");
1059  g << "if (" << flag << ") {\n" << d << ".fail = 1;\nbreak;\n}\n";
1060  }
1061  g << "break;\n";
1062  }
1063  if (step_size_ == 0 && !in_event) {
1064  g << "case IVP_STEP_CONTROL:\n";
1065  g << "casadi_ivp_control(&" << d << ");\n";
1066  g << "break;\n";
1067  }
1068  if (ne_ > 0 && !in_event) {
1069  for (casadi_int adj = 0; adj <= (nadj_ > 0 ? 1 : 0); ++adj) {
1070  g << "case IVP_EVENT" << (adj ? "_ADJ" : "") << ":\n";
1071  g << "while (casadi_ivp_event" << (adj ? "_adj" : "") << "(&" << d << ")) {\n";
1072  codegen_task(g, d, calls, true);
1073  g << "}\n";
1074  g << "break;\n";
1075  }
1076  }
1077  g << "default: " << d << ".fail = 1;\n";
1078  g << "}\n";
1079 }
