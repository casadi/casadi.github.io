6095  template<typename DataType>
6096  void Sparsity::set(DataType* data, const DataType* val_data, const Sparsity& val_sp) const {
6097  // Get dimensions of this
6098  const casadi_int sz = nnz();
6099  const casadi_int sz1 = size1();
6100  const casadi_int sz2 = size2();
6101 
6102  // Get dimensions of assigning matrix
6103  const casadi_int val_sz = val_sp.nnz();
6104  const casadi_int val_sz1 = val_sp.size1();
6105  const casadi_int val_sz2 = val_sp.size2();
6106  const casadi_int val_nel = val_sz1*val_sz2;
6107 
6108  // Check if sparsity matches
6109  if (val_sp==*this) {
6110  std::copy(val_data, val_data+sz, data);
6111  } else if (this->is_empty()) {
6112  // Quick return
6113  return;
6114  } else if (val_sp.is_empty()) {
6115  // Quick return
6116  return;
6117  } else if (val_nel==1) { // if scalar
6118  std::fill(data, data+sz, val_sz==0 ? DataType(0) : val_data[0]);
6119  } else if (sz2==val_sz2 && sz1==val_sz1) {
6120  // Matching dimensions
6121  // Sparsity
6122  const casadi_int* c = row();
6123  const casadi_int* rind = colind();
6124  const casadi_int* v_c = val_sp.row();
6125  const casadi_int* v_rind = val_sp.colind();
