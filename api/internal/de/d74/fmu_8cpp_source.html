53 Fmu::Fmu(const std::string& name, FmuApi api, const DaeBuilderInternal* dae,

453 void FmuInternal::init(const DaeBuilderInternal* dae) {
454  // Copy info from DaeBuilder
455  resource_ = dae->resource_;
456  fmutol_ = dae->fmutol_;
457  instance_name_ = dae->model_identifier_;
458  instantiation_token_ = dae->instantiation_token_;
459  logging_on_ = dae->debug_;
460  number_of_event_indicators_ = dae->nzero_;
461  provides_directional_derivatives_ = dae->provides_directional_derivatives_;
462  provides_adjoint_derivatives_ = dae->provides_adjoint_derivatives_;
463  can_be_instantiated_only_once_per_process_ = dae->can_be_instantiated_only_once_per_process_;
464  start_time_ = dae->start_time_;
465  nx_ = dae->size(Category::X);
466  do_evaluation_dance_ = dae->generation_tool_.rfind("Simulink", 0) == 0;
467 
468  // Mark input indices
469  size_t numel = 0;
470  std::vector<bool> lookup(dae->n_variables(), false);
471  for (auto&& n : scheme_in_) {
472  casadi_assert(scheme_.find(n) != scheme_.end(), "Unsupported input: '" + n + "'");
473  for (size_t i : scheme_.at(n)) {
474  casadi_assert(!lookup.at(i), "Duplicate variable: " + dae->variable(i).name);
475  lookup.at(i) = true;
476  numel++;
477  }
478  }
479  // Input mappings
480  iind_.reserve(numel);
481  iind_map_.reserve(lookup.size());
482  for (size_t k = 0; k < lookup.size(); ++k) {
483  if (lookup[k]) {
484  iind_map_.push_back(iind_.size());
485  iind_.push_back(k);
486  } else {
487  iind_map_.push_back(-1);
488  }
489  }
490  // Mark output indices
491  numel = 0;
492  std::fill(lookup.begin(), lookup.end(), false);
493  for (auto&& n : scheme_out_) {
494  casadi_assert(scheme_.find(n) != scheme_.end(), "Unsupported output: '" + n + "'");
495  for (size_t i : scheme_.at(n)) {
496  casadi_assert(!lookup.at(i), "Duplicate variable: " + dae->variable(i).name);
497  lookup.at(i) = true;
498  numel++;
499  }
500  }
501  // Construct mappings
502  oind_.reserve(numel);
503  oind_map_.reserve(lookup.size());
504  for (size_t k = 0; k < lookup.size(); ++k) {
505  if (lookup[k]) {
506  oind_map_.push_back(oind_.size());
507  oind_.push_back(k);
508  } else {
509  oind_map_.push_back(-1);
510  }
511  }
512  // Inputs
513  ired_.resize(scheme_in_.size());
514  for (size_t i = 0; i < ired_.size(); ++i) {
515  auto&& s = scheme_.at(scheme_in_[i]);
516  ired_[i].resize(s.size());
517  for (size_t k = 0; k < s.size(); ++k) {
518  ired_[i][k] = iind_map_.at(s[k]);
519  }
520  }
521  // Outputs
522  ored_.resize(scheme_out_.size());
523  for (size_t i = 0; i < ored_.size(); ++i) {
524  auto&& s = scheme_.at(scheme_out_[i]);
525  ored_[i].resize(s.size());
526  for (size_t k = 0; k < s.size(); ++k) {
527  ored_[i][k] = oind_map_.at(s[k]);
528  }
529  }
530 
531  // Is there an independent variable?
532  has_independent_ = false;
533 
534  // Collect meta information for inputs
535  nominal_in_.reserve(iind_.size());
536  min_in_.reserve(iind_.size());
537  max_in_.reserve(iind_.size());
538  vn_in_.reserve(iind_.size());
539  vr_in_.reserve(iind_.size());
540  for (size_t i : iind_) {
541  const Variable& v = dae->variable(i);
542  nominal_in_.push_back(v.nominal);
543  min_in_.push_back(v.min);
544  max_in_.push_back(v.max);
545  vn_in_.push_back(v.name);
546  vr_in_.push_back(v.value_reference);
547  if (v.causality == Causality::INDEPENDENT) {
548  if (i != 0) casadi_error("Independent variable must be the first model variable");
549  has_independent_ = true;
550  independent_vr_ = vr_in_.back();
551  }
552  }
553  // Collect meta information for outputs
554  nominal_out_.reserve(oind_.size());
555  min_out_.reserve(oind_.size());
556  max_out_.reserve(oind_.size());
557  vn_out_.reserve(oind_.size());
558  vr_out_.reserve(oind_.size());
559  for (size_t i : oind_) {
560  const Variable& v = dae->variable(i);
561  nominal_out_.push_back(v.nominal);
562  min_out_.push_back(v.min);
563  max_out_.push_back(v.max);
564  vn_out_.push_back(v.name);
565  vr_out_.push_back(v.value_reference);
566  }
567 
568  // Numerical values for inputs
569  value_in_.resize(iind_.size());
570 
571  // Get Jacobian sparsity information
572  jac_sp_ = dae->jac_sparsity(oind_, iind_);
573 
574  // Get Hessian sparsity information
575  hess_sp_ = dae->hess_sparsity(oind_, iind_);
576 
577  // Path to resource directory
578  resource_loc_ = resource_.path();
579  if (Filesystem::is_enabled()) resource_loc_ = Filesystem::absolute(resource_loc_);
580  resource_loc_ += "/resources";
581 
582  // Use forward slashes (supported by all file systems)
583  std::replace(resource_loc_.begin(), resource_loc_.end(), '\\', '/');
584 }

590  return 1;

593 void FmuInternal::finalize() {
594  warning_fired_discrete_states_need_update_ = false;
595  warning_fired_terminate_simulation_ = false;
596  warning_fired_nominals_of_continuous_states_changed_ = false;
597  warning_fired_values_of_continuous_states_changed_ = false;
598  warning_fired_next_event_time_defined_ = false;
599  // Load DLL
600  std::string instance_name_no_dot = instance_name_;
601  std::replace(instance_name_no_dot.begin(), instance_name_no_dot.end(), '.', '_');
602  std::string dll_path = resource_.path() + "/binaries/" + system_infix()
603  + "/" + instance_name_no_dot + dll_suffix();
604  li_ = Importer(dll_path, "dll");
605 
606  // Get FMI C functions
607  load_functions();
608 
609  // Create a temporary instance
610  void* c = instantiate();
611  // Set all values
612  if (set_values(c)) {
613  casadi_error("FmuInternal::set_values failed");
614  }
615  // Initialization mode begins
616  if (enter_initialization_mode(c)) {
617  casadi_error("FmuInternal::enter_initialization_mode failed");
618  }
619  // Get input values
620  if (!value_in_.empty()) {
621  // Value references
622  const unsigned int* vr = get_ptr(vr_in_);
623  size_t n_vr = vr_in_.size();
624  // Values
625  double* value = get_ptr(value_in_);
626  size_t n_value = value_in_.size();
627  // Set time variable, if any
628  if (has_independent_) {
629  *value = start_time_;
630  // Skip when getting remaining inputs
631  vr++;
632  n_vr--;
633  value++;
634  n_value--;
635  }
636  // Set remaining
637  if (n_value > 0) {
638  if (get_real(c, vr, n_vr, value, n_value)) {
639  casadi_error("FmuInternal::get_in failed");
640  }
641  }
642  }
643  // Get auxilliary variables
644  if (get_aux(c)) {
645  casadi_error("FmuInternal::get_aux failed");
646  }
647  // Free memory
648  free_instance(c);
649 }

781 Dict FmuInternal::compile_fmu(const std::string& name, const Dict& files, const Dict& opts) {
782  // Options
783  Dict compiler_opts;
784  std::string compiler_plugin = "shell";
785  std::vector<std::string> include_dirs;
786  for (auto&& op : opts) {
787  if (op.first == "compiler") {
788  compiler_plugin = op.second.to_string();
789  } else if (op.first == "compiler_options") {
790  compiler_opts = op.second;
791  } else if (op.first == "include_dirs") {
792  include_dirs = op.second.to_string_vector();
793  } else {
794  casadi_error("No such option: " + op.first);
795  }
796  }
797  // The C sources to compile are the .c entries of the export_fmu map (flat in CWD)
798  std::vector<std::string> csources;
799  for (auto&& kv : files) {
800  const std::string& local = kv.first;
801  if (local.size() > 2 && local.substr(local.size() - 2) == ".c") csources.push_back(local);
802  }
803  // Amalgamation translation unit: a single .c that #includes every generated source,
804  // because the shell Importer compiles exactly one source file. The compile runs in
805  // the working directory, so quoted includes resolve against the loose files there.
806  // This is a build scratch file, NOT FMU content, so it is removed below.
807  std::string amalg = name + "_all.c";
808  {
809  auto f = Filesystem::ofstream_ptr(amalg);
810  for (const std::string& c : csources) *f << "#include \"" << c << "\"\n";
811  }
812  // Compile the amalgamation into a shared library via the Importer plugin.
813  // Pass include dirs through the OS-agnostic 'include_dirs' option (the plugin
814  // prefixes each with the compiler's own include flag: -I on gcc/clang, /I on MSVC).
815  Dict cop = compiler_opts;
816  if (!include_dirs.empty()) cop["include_dirs"] = include_dirs;
817  Importer compiler(amalg, compiler_plugin, cop);
818  // Lift the binary to a stable path before the Importer destructor removes it, then
819  // discard the amalgamation scratch file (compilation is done once the Importer exists)
820  std::string dll = name + dll_suffix();
821  Filesystem::copy_file(compiler.library(), dll);
822  Filesystem::remove(amalg);
823  // Augment the file map with the compiled binary
824  // (FMI-3 layout: binaries/<dll_infix>/<modelIdentifier><suffix>)
825  Dict ret = files;
826  ret[dll] = "binaries/" + fmi3_dll_infix() + "/" + name + dll_suffix();
827  return ret;
828 }

948 int FmuInternal::eval_fd(FmuMemory* m, bool independent_seeds) const {
949  // Number of inputs and outputs
950  size_t n_known = m->id_in_.size();
951  size_t n_unknown = m->id_out_.size();
952  // Quick return if nothing to be calculated
953  if (n_unknown == 0) return 0;
954  // Quick return if no seeds
955  if (n_known == 0) {
956  std::fill(m->d_out_.begin(), m->d_out_.end(), 0);
957  return 0;
958  }
959  // Evalute (should not be necessary)
960  if (get_real(m->instance, get_ptr(m->vr_out_), n_unknown, get_ptr(m->v_out_), n_unknown, m)) {
961  casadi_warning("Evaluating FMU failed");
962  return 1;
963  }
964  // Make outputs dimensionless
965  for (size_t k = 0; k < n_unknown; ++k) m->v_out_[k] /= nominal_out_[m->id_out_[k]];
966  // Number of points in FD stencil
967  casadi_int n_points = n_fd_points(m->self.fd_);
968  // Offset for points
969  casadi_int offset = fd_offset(m->self.fd_);
970  // Memory for perturbed outputs
971  m->fd_out_.resize(n_points * n_unknown);
972  // Which inputs are in bounds
973  m->in_bounds_.resize(n_known);
974  // Memory for perturbed inputs
975  m->v_pert_.resize(n_known);
976  // Do any any inputs need flipping?
977  m->flip_.resize(n_known);
978  size_t first_flip = -1;
979  for (size_t i = 0; i < n_known; ++i) {
980  // Try to take step
981  double test = m->v_in_[i] + m->self.step_ * m->d_in_[i];
982  // Check if in bounds
983  size_t id = m->id_in_[i];
984  if (test >= min_in_[id] && test <= max_in_[id]) {
985  // Positive perturbation is fine
986  m->flip_[i] = false;
987  } else {
988  // Try negative direction instead
989  test = m->v_in_[i] - m->self.step_ * m->d_in_[i];
990  casadi_assert(test >= min_in_[id] && test <= max_in_[id],
991  "Cannot perturb " + vn_in_[id] + " at " + str(m->v_in_[i]) + ", min " + str(min_in_[id])
992  + ", max " + str(max_in_[id]) + ", nominal " + str(nominal_in_[id]));
993  m->flip_[i] = true;
994  if (first_flip == size_t(-1)) first_flip = i;
995  }
996  }
997  // If seeds are not independent, we have to flip the sign for all of the seeds or none
998  if (first_flip != size_t(-1) && !independent_seeds) {
999  // Flip the rest of the seeds
1000  for (size_t i = 0; i < n_known; ++i) {
1001  if (!m->flip_[i]) {
1002  // Test negative direction
1003  double test = m->v_in_[i] - m->self.step_ * m->d_in_[i];
1004  size_t id = m->id_in_[i];
1005  casadi_assert(test >= min_in_[id] && test <= max_in_[id],
1006  "Cannot perturb both " + vn_in_[id] + " and " + vn_in_[first_flip]);
1007  // Flip it too
1008  m->flip_[i] = true;
1009  }
1010  }
1011  }
1012  // All perturbed outputs
1013  const double* yk_all[5] = {nullptr};
1014 
1015  // Calculate all perturbed outputs
1016  for (casadi_int k = 0; k < n_points; ++k) {
1017  // Where to save the perturbed outputs
1018  double* yk = &m->fd_out_[n_unknown * k];
1019  casadi_assert_dev(k < 5);
1020  yk_all[k] = yk;
1021  // If unperturbed output, quick return
1022  if (k == offset) {
1023  casadi_copy(get_ptr(m->v_out_), n_unknown, yk);
1024  continue;
1025  }
1026  // Perturbation size
1027  double pert = (k - offset) * m->self.step_;
1028  // Perturb inputs, if allowed
1029  for (size_t i = 0; i < n_known; ++i) {
1030  // Try to take step
1031  double sign = m->flip_[i] ? -1 : 1;
1032  double test = m->v_in_[i] + pert * sign * m->d_in_[i];
1033  // Check if in bounds
1034  size_t id = m->id_in_[i];
1035  m->in_bounds_[i] = test >= min_in_[id] && test <= max_in_[id];
1036  // Take step, if allowed
1037  m->v_pert_[i] = m->in_bounds_[i] ? test : m->v_in_[i];
1038  }
1039  // Pass perturbed inputs to FMU
1040  if (set_all(m, get_ptr(m->v_pert_), m->v_pert_.size())) return 1;
1041  // Evaluate perturbed FMU
1042  if (get_all(m, yk, n_unknown)) return 1;
1043  // Post-process yk if there was any scaling
1044  if (independent_seeds) {
1045  for (size_t i = 0; i < n_unknown; ++i) {
1046  // Variable id
1047  size_t id = m->id_out_[i];
1048  // Differentiation with respect to what variable
1049  size_t wrt_id = m->wrt_.at(id);
1050  // Find the corresponding input variable
1051  size_t wrt_i;
1052  for (wrt_i = 0; wrt_i < n_known; ++wrt_i) {
1053  if (m->id_in_[wrt_i] == wrt_id) break;
1054  }
1055  // Check if in bounds
1056  if (m->in_bounds_.at(wrt_i) && (m->self.fd_flip_ || !m->flip_[wrt_i])) {
1057  // Input was in bounds: Keep output, make dimensionless
1058  yk[i] /= nominal_out_[m->id_out_[i]];
1059  } else {
1060  // Input was out of bounds: Discard output
1061  yk[i] = nan;
1062  }
1063  }
1064  }
1065  }
1066  // Restore FMU inputs
1067  if (set_all(m, get_ptr(m->v_in_), m->v_in_.size())) return 1;
1068  // Step size
1069  double h = m->self.step_;
1070 
1071  // Calculate FD approximation
1072  finite_diff(m->self.fd_, yk_all, get_ptr(m->d_out_), h, n_unknown, eps);
1073 
1074  // If seeds are dependent, quick return
1075  if (!independent_seeds) {
1076  // Simpy copy the results to output (no validation)
1077  for (size_t ind = 0; ind < m->id_out_.size(); ++ind) {
1078  m->osens_[m->id_out_[ind]] = m->d_out_[ind];
1079  }
1080  return 0;
1081  }
1082 
1083  // Collect requested variables
1084  for (size_t ind = 0; ind < m->id_out_.size(); ++ind) {
1085  // Variable id
1086  size_t id = m->id_out_[ind];
1087  // With respect to what variable
1088  size_t wrt = m->wrt_[id];
1089  // Find the corresponding input variable
1090  size_t wrt_i;
1091  for (wrt_i = 0; wrt_i < n_known; ++wrt_i) {
1092  if (m->id_in_[wrt_i] == wrt) break;
1093  }
1094  // Nominal value
1095  double n = nominal_out_[id];
1096  // Get the value
1097  double d_fd = m->d_out_[ind] * n;
1098  // Correct sign, if necessary
1099  if (m->flip_[wrt_i]) d_fd = -d_fd;
1100  // Use FD instead of AD or to compare with AD
1101  if (m->self.validate_forward_) {
1102  // Value to compare with
1103  double d_ad = m->osens_[id];
1104  // Nominal value used as seed
1105  d_ad /= nominal_in_[wrt];
1106  d_fd /= nominal_in_[wrt];
1107  // Is it a not a number?
1108  bool d_is_nan = d_ad != d_ad;
1109  // Magnitude of derivatives
1110  double d_max = std::fmax(std::fabs(d_fd), std::fabs(d_ad));
1111  // Check if NaN or error exceeds thresholds
1112  if (d_is_nan || (d_max > n * m->self.abstol_
1113  && std::fabs(d_ad - d_fd) > d_max * m->self.reltol_)) {
1114  // Offset for printing the stencil
1115  double off = m->fd_out_.at(ind + offset * n_unknown);
1116  // Warning or add to file
1117  std::stringstream ss;
1118  if (m->self.validate_ad_file_.empty()) {
1119  // Issue warning
1120  ss << (d_is_nan ? "NaN" : "Inconsistent") << " derivatives of " << vn_out_[id]
1121  << " w.r.t. " << desc_in(m, wrt) << ", got " << d_ad
1122  << " for AD vs. " << d_fd << " for FD[" << to_string(m->self.fd_) << "].";
1123  // Print the stencil:
1124  ss << "\nValues for step size " << h << ": " << (n * off) << " + [";
1125  for (casadi_int k = 0; k < n_points; ++k) {
1126  if (k > 0) ss << ", ";
1127  ss << (n * (m->fd_out_.at(ind + k * n_unknown) - off));
1128  }
1129  ss << "]";
1130  // Issue warning
1131  casadi_warning(ss.str());
1132  } else {
1133  // Output
1134  ss << vn_out_[id] << " ";
1135  // Input
1136  ss << vn_in_[wrt] << " ";
1137  // Value
1138  ss << m->ibuf_[wrt] << " ";
1139  // Noninal
1140  ss << nominal_in_[wrt] << " ";
1141  // Min
1142  ss << min_in_[wrt] << " ";
1143  // Max
1144  ss << max_in_[wrt] << " ";
1145  // AD
1146  ss << d_ad << " ";
1147  // FD
1148  ss << d_fd << " ";
1149  // Step
1150  ss << h << " ";
1151  // Offset
1152  ss << off << " ";
1153  // Stencil
1154  ss << "[";
1155  for (casadi_int k = 0; k < n_points; ++k) {
1156  if (k > 0) ss << ",";

1159  ss << "]" << std::endl;
1160  // Append to file
